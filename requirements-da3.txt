#   .venv\Scripts\python.exe -m pip install --no-deps depth-anything-3==0.1.1
#   .venv\Scripts\python.exe -m pip install -r requirements-da3.txt
#
# torchvision must match the installed torch; pip picks the matching build and
# leaves torch itself untouched.
#
# NOTE: keep this file ASCII-only. On Windows pip may decode requirements files
# with the locale codec (GBK), and non-ASCII comments make it fail.
